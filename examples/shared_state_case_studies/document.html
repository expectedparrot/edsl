1. The machine

What it stores

SharedDocument has 2 durable fields: text, revisions.

Its public view exposes title, text, revision_count, recent_revisions.

How it changes

Survey interviews can issue revise. Inputs are validated and each accepted command is applied atomically.

Completion predicate: none. Close effects: 0.

2. Complete, standalone Python

This is one copyable program: it defines the machine, creates the agents and Survey, selects the schedule, runs Gemini locally, and prints the Results. It uses only public EDSL imports—there are no repository-local imports or private helper functions.

from uuid import uuid4
from edsl import (
    Agent,
    AgentList,
    InterviewSchedule,
    Model,
    QuestionFreeText,
    QuestionMultipleChoice,
    QuestionNumerical,
    QuestionRank,
    Survey,
)
from edsl.sharedstate import (
    Command,
    Machine,
    T,
    append,
    constant,
    field,
    input_,
    record,
    reduce_,
    set_,
    state_field,
    SharedState,
    SharedStateMap,
    current,
)

document_machine = Machine(
    name="SharedDocument",
    constants={
        "title": "Climate cooperation translation chain",
        "initial_text": "The atmosphere does not recognize national borders. Each country benefits when others reduce emissions, yet each also faces a temptation to delay its own costly action. Durable cooperation therefore requires credible commitments, transparent measurement, and a fair distribution of costs. Wealthier societies have greater capacity to finance the transition, while poorer societies often face the gravest immediate risks. A successful agreement must align individual incentives with the shared interest in a stable climate.",
    },
    fields={
        "text": state_field(T.text(), constant("initial_text")),
        "revisions": state_field(T.sequence(T.map()), []),
    },
    commands={
        "revise": Command(
            inputs={
                "author": T.text(),
                "round": T.integer(minimum=1),
                "text": T.text(),
                "rationale": T.text(),
            },
            effects=(
                set_("text", input_("text")),
                append(
                    "revisions",
                    record(
                        author=input_("author"),
                        round=input_("round"),
                        rationale=input_("rationale"),
                        changed=input_("text") != field("text"),
                    ),
                ),
            ),
        )
    },
    view={
        "title": constant("title"),
        "text": field("text"),
        "revision_count": field("revisions").length(),
        "recent_revisions": reduce_("tail", field("revisions"), count=10),
    },
)


def build_experiment():
    states = SharedStateMap(
        SharedState(game=document_machine), state_id=f"gemini-document-{uuid4()}"
    )
    game = states.by("plan").game
    text = QuestionFreeText(
        question_name="text",
        question_text="Translate the complete current document into {{ agent.target_language }}. Preserve its meaning, tone, and five-sentence structure. Return only the translation, without commentary.\n\nCurrent document:\n{{ shared_state.game.text }}",
    )
    survey = Survey(
        [
            game.read(),
            text,
            game.revise(
                author=current.agent.name,
                round=current.agent.turn,
                text=text.answer,
                rationale="Translate the current document",
            ),
        ]
    )
    people = AgentList(
        [
            Agent(
                name="Spanish translator",
                traits={"plan_id": "climate", "turn": 1, "target_language": "Spanish"},
            ),
            Agent(
                name="French translator",
                traits={"plan_id": "climate", "turn": 2, "target_language": "French"},
            ),
            Agent(
                name="German translator",
                traits={"plan_id": "climate", "turn": 3, "target_language": "German"},
            ),
            Agent(
                name="Japanese translator",
                traits={"plan_id": "climate", "turn": 4, "target_language": "Japanese"},
            ),
            Agent(
                name="Swahili translator",
                traits={"plan_id": "climate", "turn": 5, "target_language": "Swahili"},
            ),
            Agent(
                name="English translator",
                traits={"plan_id": "climate", "turn": 6, "target_language": "English"},
            ),
        ]
    )
    return (survey, people, InterviewSchedule.grouped_round_robin("plan_id", "turn"))


if __name__ == "__main__":
    survey, agents, schedule = build_experiment()
    results = (
        survey.by(agents)
        .by(Model("gemini-2.5-flash", service_name="google"))
        .run(
            cache=False,
            disable_remote_cache=True,
            disable_remote_inference=True,
            interview_schedule=schedule,
            max_concurrency=5,
        )
    )
    print(results)

Observed result

Six translators passed the document from English through Spanish, French, German, Japanese, and Swahili, then back to English. The final round-trip text was: “The atmosphere knows no boundaries. Every country benefits from emissions reductions made by other countries, but at the same time is tempted to delay its own costly actions. Therefore, sustained cooperation requires credible commitments, transparent actions, and an equitable sharing of costs. While richer societies have greater capacity to finance the transition, poorer societies often face the greatest immediate risks. Successful agreements must balance individual incentives with the collective benefits of a stable climate.”
