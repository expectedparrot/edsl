This page connects one concrete social-science experiment to the state machine that enforced it and the real Gemini responses that changed its state.

1. The machine

What it stores

SharedForecast has 1 durable field: forecasts.

Its public view exposes latest, history, mean_probability, median_probability, confidence_weighted_probability.

How it changes

Survey interviews can issue submit. Inputs are validated and each accepted command is applied atomically.

Completion predicate: none. Close effects: 0.

2. Complete, standalone Python

This is one copyable program: it defines the machine, creates the agents and Survey, selects the schedule, runs Gemini locally, and prints the Results. It uses only public EDSL imports—there are no repository-local imports or private helper functions.

from uuid import uuid4
from edsl import (
    Agent,
    AgentList,
    InterviewSchedule,
    Model,
    QuestionFreeText,
    QuestionMultipleChoice,
    QuestionNumerical,
    QuestionRank,
    Survey,
)
from edsl.sharedstate import (
    Command,
    Machine,
    T,
    append,
    choose,
    current,
    field,
    input_,
    local,
    map_sequence,
    record,
    reduce_,
    state_field,
    SharedState,
    SharedStateMap,
)

latest_map = reduce_("latest_by", field("forecasts"), field="forecaster")
latest_records = latest_map.values()
public_latest = map_sequence(
    latest_records,
    item="forecast",
    value_expr=record(
        forecaster=local("forecast").get("forecaster"),
        round=local("forecast").get("round"),
        probability=local("forecast").get("probability"),
        confidence=local("forecast").get("confidence"),
    ),
)
public_history = map_sequence(
    field("forecasts"),
    item="forecast",
    value_expr=record(
        forecaster=local("forecast").get("forecaster"),
        round=local("forecast").get("round"),
        probability=local("forecast").get("probability"),
        confidence=local("forecast").get("confidence"),
    ),
)
probabilities = map_sequence(
    latest_records, item="forecast", value_expr=local("forecast").get("probability")
)
confidences = map_sequence(
    latest_records, item="forecast", value_expr=local("forecast").get("confidence")
)
weighted_terms = map_sequence(
    latest_records,
    item="forecast",
    value_expr=local("forecast").get("probability")
    * local("forecast").get("confidence"),
)
denominator = reduce_("sum", confidences)
forecast_machine = Machine(
    name="SharedForecast",
    constants={},
    fields={"forecasts": state_field(T.sequence(T.map()), [])},
    commands={
        "submit": Command(
            inputs={
                "forecaster": T.text(),
                "round": T.integer(minimum=1),
                "probability": T.number(minimum=0, maximum=100),
                "confidence": T.number(minimum=0, maximum=100),
            },
            effects=(
                append(
                    "forecasts",
                    record(
                        forecaster=input_("forecaster"),
                        round=input_("round"),
                        probability=input_("probability"),
                        confidence=input_("confidence"),
                        interview=current("interview_id"),
                    ),
                ),
            ),
        )
    },
    view={
        "latest": public_latest,
        "history": public_history,
        "mean_probability": reduce_("mean", probabilities),
        "median_probability": reduce_("median", probabilities),
        "confidence_weighted_probability": choose(
            denominator > 0, reduce_("sum", weighted_terms) / denominator, None
        ),
    },
)


def build_experiment():
    states = SharedStateMap(
        SharedState(game=forecast_machine), state_id=f"gemini-forecast-{uuid4()}"
    )
    game = states.by("panel").game
    probability = QuestionNumerical(
        question_name="probability",
        question_text="Estimate the probability (0–100) that the event occurs. Your private initial belief is {{ agent.initial }}. Current consensus: {{ shared_state.game }}.",
        min_value=0,
        max_value=100,
    )
    survey = Survey(
        [
            game.read(),
            probability,
            game.submit(
                forecaster=current.agent.name,
                round=current.run.round,
                probability=probability.answer,
                confidence=current.agent.confidence,
            ),
        ]
    )
    people = AgentList(
        [
            Agent(
                name="Forecaster A",
                traits={"panel_id": "panel", "initial": 30, "confidence": 75},
            ),
            Agent(
                name="Forecaster B",
                traits={"panel_id": "panel", "initial": 50, "confidence": 80},
            ),
            Agent(
                name="Forecaster C",
                traits={"panel_id": "panel", "initial": 70, "confidence": 70},
            ),
        ]
    )
    return (
        survey,
        people,
        InterviewSchedule.rounds(
            count=2,
            group_by="panel_id",
            within_round="concurrent",
            state_visibility="snapshot",
        ),
    )


if __name__ == "__main__":
    survey, agents, schedule = build_experiment()
    results = (
        survey.by(agents)
        .by(Model("gemini-2.5-flash", service_name="google"))
        .run(
            cache=False,
            disable_remote_cache=True,
            disable_remote_inference=True,
            interview_schedule=schedule,
            max_concurrency=5,
        )
    )
    print(results)
