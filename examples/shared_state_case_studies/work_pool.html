1. The machine

What it stores

SharedWorkPool has 3 durable fields: available, claims, completed.

Its public view exposes available, claims, completed.

How it changes

Survey interviews can issue claim_before, complete. Inputs are validated and each accepted command is applied atomically.

Completion predicate: none. Close effects: 0.

2. Complete, standalone Python

This is one copyable program: it defines the machine, creates the agents and Survey, selects the schedule, runs Gemini locally, and prints the Results. It uses only public EDSL imports—there are no repository-local imports or private helper functions.

from uuid import uuid4
from edsl import (
    Agent,
    AgentList,
    InterviewSchedule,
    Model,
    QuestionFreeText,
    QuestionMultipleChoice,
    QuestionNumerical,
    QuestionRank,
    Survey,
)
from edsl.sharedstate import (
    Command,
    Machine,
    T,
    constant,
    field,
    input_,
    put,
    record,
    set_,
    state_field,
    SharedState,
    SharedStateMap,
    current,
)

unclaimed = ~field("claims").contains(input_("claimant"))
work_pool_machine = Machine(
    name="SharedWorkPool",
    constants={"items": ({"id": "W1"}, {"id": "W2"})},
    fields={
        "available": state_field(T.sequence(), constant("items")),
        "claims": state_field(T.map(), {}),
        "completed": state_field(T.map(), {}),
    },
    commands={
        "claim_before": Command(
            inputs={"claimant": T.text()},
            require=unclaimed,
            effects=(
                put("claims", input_("claimant"), field("available").first()),
                set_("available", field("available").drop_first()),
            ),
            timing="before_question",
        ),
        "complete": Command(
            inputs={"claimant": T.text(), "result": T.any()},
            require=field("claims").contains(input_("claimant")),
            effects=(
                put(
                    "completed",
                    input_("claimant"),
                    record(
                        item=field("claims").get(input_("claimant")),
                        result=input_("result"),
                    ),
                ),
            ),
        ),
    },
    view={
        "available": field("available"),
        "claims": field("claims"),
        "completed": field("completed"),
    },
)


def build_experiment():
    states = SharedStateMap(
        SharedState(game=work_pool_machine), state_id=f"gemini-work-{uuid4()}"
    )
    game = states.by("queue").game
    result = QuestionFreeText(
        question_name="result",
        question_text="You atomically claimed work item {{ shared_state.game.claims }}. Return a short completion note for your item.",
    )
    survey = Survey(
        [
            game.claim_before(claimant=current.agent.name),
            game.read(),
            result,
            game.complete(claimant=current.agent.name, result={"note": result.answer}),
        ]
    )
    return (
        survey,
        AgentList(
            [Agent(name="Worker A", traits={}), Agent(name="Worker B", traits={})]
        ),
        InterviewSchedule.rounds(
            count=1, within_round="concurrent", state_visibility="live"
        ),
    )


if __name__ == "__main__":
    survey, agents, schedule = build_experiment()
    results = (
        survey.by(agents)
        .by(Model("gemini-2.5-flash", service_name="google"))
        .run(
            cache=False,
            disable_remote_cache=True,
            disable_remote_inference=True,
            interview_schedule=schedule,
            max_concurrency=5,
        )
    )
    print(results)

6. What this example teaches us

Before-question atomic claims work under live visibility. The assignment order is completion-dependent, which is correct but should be made explicit when reproducibility matters.

This is one behavioral realization, not an estimate of Gemini's average behavior. Its main purpose is to test whether the machine, survey, visibility, and schedule jointly express the intended experiment.
