1. The machine

What it stores

SharedAgenda has 2 durable fields: proposals, ballots.

Its public view exposes proposals, ballots, scores.

How it changes

Survey interviews can issue propose, vote. Inputs are validated and each accepted command is applied atomically.

Completion predicate: none. Close effects: 0.

2. Complete, standalone Python

This is one copyable program: it defines the machine, creates the agents and Survey, selects the schedule, runs Gemini locally, and prints the Results. It uses only public EDSL imports—there are no repository-local imports or private helper functions.

from uuid import uuid4
from edsl import (
    Agent,
    AgentList,
    InterviewSchedule,
    Model,
    QuestionFreeText,
    QuestionMultipleChoice,
    QuestionNumerical,
    QuestionRank,
    Survey,
)
from edsl.sharedstate import (
    Command,
    Machine,
    T,
    append,
    decode_matrix,
    expr,
    field,
    input_,
    local,
    map_sequence,
    record,
    reduce_,
    state_field,
    SharedState,
    SharedStateMap,
    current,
)

proposal_id = expr("concat", "A", field("proposals").length() + 1)
proposal_titles = map_sequence(
    field("proposals"), item="proposal", value_expr=local("proposal").get("title")
)
vote_options = ["up", "neutral", "down"]
agenda_machine = Machine(
    name="SharedAgenda",
    constants={"vote_weights": {"up": 1, "neutral": 0, "down": -1}},
    fields={
        "proposals": state_field(T.sequence(), []),
        "ballots": state_field(T.sequence(), []),
    },
    commands={
        "propose": Command(
            inputs={"proposer": T.text(), "title": T.text()},
            effects=(
                append(
                    "proposals",
                    record(
                        id=proposal_id,
                        proposer=input_("proposer"),
                        title=input_("title"),
                    ),
                ),
            ),
        ),
        "vote": Command(
            inputs={"voter": T.text(), "votes": T.map()},
            effects=(
                append(
                    "ballots",
                    record(
                        voter=input_("voter"),
                        votes=decode_matrix(
                            input_("votes"), rows=proposal_titles, options=vote_options
                        ),
                    ),
                ),
            ),
        ),
    },
    view={
        "proposals": field("proposals"),
        "ballots": field("ballots"),
        "scores": reduce_(
            "weighted_matrix_tally",
            field("ballots"),
            weights={"up": 1, "neutral": 0, "down": -1},
        ),
    },
)


def build_experiment():
    states = SharedStateMap(
        SharedState(game=agenda_machine), state_id=f"gemini-agenda-{uuid4()}"
    )
    game = states.by("committee").game
    title = QuestionFreeText(
        question_name="title",
        question_text="Propose one concise weekend activity for the committee's agenda. Your proposal should reflect this priority: {{ agent.priority }}. Return only the activity title.",
    )
    ballot = QuestionMatrix(
        question_name="ballot",
        question_text="Vote on every activity proposed by the committee. Consider your own priority ({{ agent.priority }}) as well as what would work for the group.",
        question_items="{{ shared_state.game.proposals | map(attribute='title') | list }}",
        question_options=["up", "neutral", "down"],
        answering_instructions='Return one JSON object on the first line, mapping every row code to one column code, for example {"0": 0, "1": 2}. Include every row. Do not use a Markdown code fence. You may explain the votes on the next line.',
    )
    survey = Survey(
        [
            game.read(),
            title,
            game.propose(proposer=current.agent.participant, title=title.answer),
            game.read(),
            ballot,
            game.vote(voter=current.agent.participant, votes=ballot.answer),
        ]
    )
    survey.add_skip_rule("title", "'{{ agent.phase }}' != 'propose'")
    survey.add_skip_rule("ballot", "'{{ agent.phase }}' != 'vote'")
    participant_profiles = [
        ("Avery", "an accessible outdoor activity with light exercise"),
        ("Blair", "an inexpensive indoor activity that works in bad weather"),
        ("Casey", "accessibility and broad participation"),
        ("Devon", "novelty and opportunities for conversation"),
    ]
    people = AgentList(
        [
            Agent(
                name=f"{participant} — proposal",
                traits={
                    "participant": participant,
                    "phase": "propose",
                    "committee_id": "c",
                    "turn": index,
                    "priority": priority,
                },
            )
            for index, (participant, priority) in enumerate(participant_profiles)
        ]
        + [
            Agent(
                name=f"{participant} — ballot",
                traits={
                    "participant": participant,
                    "phase": "vote",
                    "committee_id": "c",
                    "turn": index + len(participant_profiles),
                    "priority": priority,
                },
            )
            for index, (participant, priority) in enumerate(participant_profiles)
        ]
    )
    return (
        survey,
        people,
        InterviewSchedule.grouped_round_robin("committee_id", "turn"),
    )


if __name__ == "__main__":
    survey, agents, schedule = build_experiment()
    results = (
        survey.by(agents)
        .by(Model("gemini-2.5-flash", service_name="google"))
        .run(
            cache=False,
            disable_remote_cache=True,
            disable_remote_inference=True,
            interview_schedule=schedule,
            max_concurrency=5,
        )
    )
    print(results)
