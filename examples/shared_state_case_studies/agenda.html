1. The machine

What it stores

SharedAgenda has 2 durable fields: proposals, ballots.

Its public view exposes proposals, ballots, scores.

How it changes

Survey interviews can issue propose, vote. Inputs are validated and each accepted command is applied atomically.

Completion predicate: none. Close effects: 0.

2. Complete, standalone Python

This is one copyable program: it defines the machine, creates the agents and Survey, selects the schedule, runs Gemini locally, and prints the Results. It uses only public EDSL imports—there are no repository-local imports or private helper functions.

from uuid import uuid4
from edsl import (
    Agent,
    AgentList,
    InterviewSchedule,
    Model,
    QuestionFreeText,
    QuestionMultipleChoice,
    QuestionNumerical,
    QuestionRank,
    Survey,
)
from edsl.sharedstate import (
    Command,
    Machine,
    T,
    append,
    expr,
    field,
    input_,
    record,
    reduce_,
    state_field,
    SharedState,
    SharedStateMap,
    current,
)

proposal_id = expr("concat", "A", field("proposals").length() + 1)
agenda_machine = Machine(
    name="SharedAgenda",
    constants={"vote_weights": {"up": 1, "neutral": 0, "down": -1}},
    fields={
        "proposals": state_field(T.sequence(), []),
        "ballots": state_field(T.sequence(), []),
    },
    commands={
        "propose": Command(
            inputs={"proposer": T.text(), "title": T.text()},
            effects=(
                append(
                    "proposals",
                    record(
                        id=proposal_id,
                        proposer=input_("proposer"),
                        title=input_("title"),
                    ),
                ),
            ),
        ),
        "vote": Command(
            inputs={
                "voter": T.text(),
                "votes": T.map(T.text(), T.choice(("up", "neutral", "down"))),
            },
            effects=(
                append("ballots", record(voter=input_("voter"), votes=input_("votes"))),
            ),
        ),
    },
    view={
        "proposals": field("proposals"),
        "ballots": field("ballots"),
        "scores": reduce_(
            "weighted_matrix_tally",
            field("ballots"),
            weights={"up": 1, "neutral": 0, "down": -1},
        ),
    },
)


def build_experiment():
    states = SharedStateMap(
        SharedState(game=agenda_machine), state_id=f"gemini-agenda-{uuid4()}"
    )
    game = states.by("committee").game
    title = QuestionFreeText(
        question_name="title",
        question_text="Propose one concise weekend activity for the committee's agenda.",
    )
    vote_a1 = QuestionMultipleChoice(
        question_name="vote_a1",
        question_text="Current agenda: {{ shared_state.game.proposals }}. Vote up, neutral, or down on proposal A1.",
        question_options=["up", "neutral", "down"],
    )
    vote_a2 = QuestionMultipleChoice(
        question_name="vote_a2",
        question_text="Vote up, neutral, or down on proposal A2.",
        question_options=["up", "neutral", "down"],
    )
    survey = Survey(
        [
            game.read(),
            title,
            game.propose(proposer=current.agent.name, title=title.answer),
            game.read(),
            vote_a1,
            vote_a2,
            game.vote(
                voter=current.agent.name,
                votes={"A1": vote_a1.answer, "A2": vote_a2.answer},
            ),
        ]
    )
    survey.add_skip_rule("title", "'{{ agent.role }}' != 'proposer'")
    survey.add_skip_rule("vote_a1", "'{{ agent.role }}' != 'voter'")
    survey.add_skip_rule("vote_a2", "'{{ agent.role }}' != 'voter'")
    people = AgentList(
        [
            Agent(
                name="Proposer 1",
                traits={"role": "proposer", "committee_id": "c", "turn": 0},
            ),
            Agent(
                name="Proposer 2",
                traits={"role": "proposer", "committee_id": "c", "turn": 1},
            ),
            Agent(
                name="Voter 1", traits={"role": "voter", "committee_id": "c", "turn": 2}
            ),
            Agent(
                name="Voter 2", traits={"role": "voter", "committee_id": "c", "turn": 3}
            ),
        ]
    )
    return (
        survey,
        people,
        InterviewSchedule.grouped_round_robin("committee_id", "turn"),
    )


if __name__ == "__main__":
    survey, agents, schedule = build_experiment()
    results = (
        survey.by(agents)
        .by(Model("gemini-2.5-flash", service_name="google"))
        .run(
            cache=False,
            disable_remote_cache=True,
            disable_remote_inference=True,
            interview_schedule=schedule,
            max_concurrency=5,
        )
    )
    print(results)

The recorded read versions were [0, 1, 2, 3]. The committed command sequence was ['propose', 'propose', 'vote', 'vote'].

How to read this: agents sharing a version decided from the same snapshot; increasing versions mean later interviews could observe earlier commits. Command order is commit order, not necessarily interview launch order.

6. What this example teaches us

A Survey can express distinct proposal and voting phases with sequential visibility. Raw Results snapshots omit the derived score view, which is an analysis limitation.

This is one behavioral realization, not an estimate of Gemini's average behavior. Its main purpose is to test whether the machine, survey, visibility, and schedule jointly express the intended experiment.
