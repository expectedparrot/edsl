1. The machine

What it stores

SharedUltimatumGame has 4 durable fields: offer, proposer, responder, decision.

Its public view exposes offer, decision, payoffs.

How it changes

Survey interviews can issue offer, respond. Inputs are validated and each accepted command is applied atomically.

Completion predicate: defined. Close effects: 0.

2. Complete, standalone Python

This is one copyable program: it defines the machine, creates the agents and Survey, selects the schedule, runs Gemini locally, and prints the Results. It uses only public EDSL imports—there are no repository-local imports or private helper functions.

from uuid import uuid4
from edsl import (
    Agent,
    AgentList,
    InterviewSchedule,
    Model,
    QuestionFreeText,
    QuestionMultipleChoice,
    QuestionNumerical,
    QuestionRank,
    Survey,
)
from edsl.sharedstate import (
    Command,
    Machine,
    T,
    choose,
    constant,
    field,
    input_,
    record,
    set_once,
    state_field,
    SharedState,
    SharedStateMap,
    current,
)

ultimatum_machine = Machine(
    name="SharedUltimatumGame",
    constants={"stake": 100},
    fields={
        "offer": state_field(T.optional(T.number()), None),
        "proposer": state_field(T.optional(T.text()), None),
        "responder": state_field(T.optional(T.text()), None),
        "decision": state_field(T.optional(T.choice(("accept", "reject"))), None),
    },
    commands={
        "offer": Command(
            inputs={
                "player": T.text(),
                "amount": T.number(minimum=0, maximum=constant("stake")),
            },
            effects=(
                set_once("proposer", input_("player")),
                set_once("offer", input_("amount")),
            ),
        ),
        "respond": Command(
            inputs={"player": T.text(), "decision": T.choice(("accept", "reject"))},
            require=field("offer") != None,
            effects=(
                set_once("responder", input_("player")),
                set_once("decision", input_("decision")),
            ),
        ),
    },
    view={
        "offer": field("offer"),
        "decision": field("decision"),
        "payoffs": choose(
            field("decision") == "accept",
            record(
                proposer=constant("stake") - field("offer"), responder=field("offer")
            ),
            record(proposer=0, responder=0),
        ),
    },
    complete_when=field("decision") != None,
)


def build_experiment():
    states = SharedStateMap(
        SharedState(game=ultimatum_machine), state_id=f"gemini-ultimatum-{uuid4()}"
    )
    game = states.by(current.agent.pair_id).game
    offer = QuestionNumerical(
        question_name="offer",
        question_text="You are the proposer. Current game: {{ shared_state.game }}. How many dollars from the $100 stake do you offer the responder?",
        min_value=0,
        max_value=100,
    )
    decision = QuestionMultipleChoice(
        question_name="decision",
        question_text="You are the responder. Current game: {{ shared_state.game }}. Do you accept or reject the recorded offer?",
        question_options=["accept", "reject"],
    )
    survey = Survey(
        [
            game.read(),
            offer,
            game.offer(player=current.agent.name, amount=offer.answer),
            game.read(),
            decision,
            game.respond(player=current.agent.name, decision=decision.answer),
        ]
    )
    survey.add_skip_rule("offer", "'{{ agent.role }}' != 'proposer'")
    survey.add_skip_rule("decision", "'{{ agent.role }}' != 'responder'")
    people = AgentList(
        [
            Agent(
                name="Proposer",
                traits={
                    "role": "proposer",
                    "pair_id": "game",
                    "turn": 0,
                    "generosity": 0.2,
                    "inequity_aversion": 0.4,
                },
            ),
            Agent(
                name="Responder",
                traits={
                    "role": "responder",
                    "pair_id": "game",
                    "turn": 1,
                    "generosity": 0.0,
                    "inequity_aversion": 0.8,
                },
            ),
        ],
        traits_presentation_template="Your traits range from -1 to 1. generosity measures willingness to sacrifice payoff for another person; inequity_aversion measures opposition to unequal outcomes. Act consistently with these values.",
    )
    return (
        survey,
        people,
        InterviewSchedule.grouped_round_robin(
            "pair_id", "turn", finalize_when=game.is_complete()
        ),
    )


if __name__ == "__main__":
    survey, agents, schedule = build_experiment()
    results = (
        survey.by(agents)
        .by(Model("gemini-2.5-flash", service_name="google"))
        .run(
            cache=False,
            disable_remote_cache=True,
            disable_remote_inference=True,
            interview_schedule=schedule,
            max_concurrency=5,
        )
    )
    print(results)

5. State evolution

The recorded read versions were [0, 1]. The committed command sequence was ['offer', 'respond', '$close'].

How to read this: agents sharing a version decided from the same snapshot; increasing versions mean later interviews could observe earlier commits. Command order is commit order, not necessarily interview launch order.

6. What this example teaches us

Ordered roles and just-in-time state refresh worked: the responder read version 1 containing the actual offer.

This is one behavioral realization, not an estimate of Gemini's average behavior. Its main purpose is to test whether the machine, survey, visibility, and schedule jointly express the intended experiment.
